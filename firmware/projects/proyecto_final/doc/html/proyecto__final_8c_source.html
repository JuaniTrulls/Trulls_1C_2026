133void FuncInterruptEncoderCLK(void* param) {
134 BaseType_t xHigherPriorityTaskWoken = pdFALSE;
135
136 if (estado_actual == ESTADO_CONFIGURADO) {
137 bool dt_state = GPIORead(ENCODER_DT);
138
139 if (dt_state == false) {
140 xTaskNotifyFromISR(encoder_task_handle, GIRO_HORARIO, eSetBits, &xHigherPriorityTaskWoken);
141 } else {
142 xTaskNotifyFromISR(encoder_task_handle, GIRO_ANTIHORARIO, eSetBits, &xHigherPriorityTaskWoken);
143 }
144
145 if (xHigherPriorityTaskWoken) {
146 portYIELD_FROM_ISR();
147 }
148 }

177static void vEncoderTask(void* pvParameter) {
178 uint32_t notificacion_recibida;
179 int8_t acumulador_pasos = 0;
180
181 while (true) {
182 xTaskNotifyWait(0x00, ULONG_MAX, &notificacion_recibida, portMAX_DELAY);
183
184 if (estado_actual == ESTADO_CONFIGURADO) {
185 if (notificacion_recibida & GIRO_HORARIO) {
186 acumulador_pasos++;
187 if (acumulador_pasos >= PASOS_POR_MINUTO) {
188 acumulador_pasos = 0;
189 if (tiempo_seteado_min < 999) {
190 tiempo_seteado_min++;
191 }
192 }
193 }
194 else if (notificacion_recibida & GIRO_ANTIHORARIO) {
195 acumulador_pasos--;
196 if (acumulador_pasos <= -PASOS_POR_MINUTO) {
197 acumulador_pasos = 0;
198 if (tiempo_seteado_min > 0) {
199 tiempo_seteado_min--;
200 }
201 }
202 }
203 }
204 vTaskDelay(pdMS_TO_TICKS(10));
205 }
206}

216static void vControlTask(void* pvParameter) {
217 GPIOInit(ENCODER_SW, GPIO_INPUT);
218 GPIOInit(ENCODER_CLK, GPIO_INPUT);
219 GPIOInit(ENCODER_DT, GPIO_INPUT);
220
221 GPIOActivInt(ENCODER_CLK, FuncInterruptEncoderCLK, false, NULL);
222
223 // Estado inicial: LEDs en configuración y el Servo arranca abierto
224 LedOn(LED_1);
225 LedOff(LED_2);
226 ServoMove(SERVO_ACCIONADOR, ANGULO_ABIERTO);
227
228 while (true) {
229 // --- MODO CONFIGURACIÓN ---
230 if (estado_actual == ESTADO_CONFIGURADO) {
231 if (GPIORead(ENCODER_SW) == 0) {
232 vTaskDelay(pdMS_TO_TICKS(50));
233 if (GPIORead(ENCODER_SW) == 0 && tiempo_seteado_min > 0) {
234
235 tiempo_restante_seg = tiempo_seteado_min * 60;
236 estado_actual = ESTADO_BLOQUEADO;
237
238 // Conmutación de indicadores visuales
239 LedOff(LED_1);
240 LedOn(LED_2);
241
242 // ACCIÓN DEL SERVO: Gira a la posición de trabado en GPIO_6
243 ServoMove(SERVO_ACCIONADOR, ANGULO_CERRADO);
244 }
245 }
246 }
247
248 // --- MODO TEMPORIZADOR EN CURSO (BLOQUEADO E ININTERRUMPIBLE) ---
249 if (estado_actual == ESTADO_BLOQUEADO) {
250 if (ulTaskNotifyTake(pdTRUE, pdMS_TO_TICKS(100)) > 0) {
251
252 tiempo_seteado_min = 0;
253 tiempo_restante_seg = 0;
254 estado_actual = ESTADO_CONFIGURADO;
255
256 // Restablece el estado visual de los LEDs
257 LedOff(LED_2);
258 LedOn(LED_1);
259
260 // ACCIÓN DEL SERVO: Gira a la posición original abriendo la caja
261 ServoMove(SERVO_ACCIONADOR, ANGULO_ABIERTO);
262 }
263 }
264 vTaskDelay(pdMS_TO_TICKS(20));
265 }

275static void vDisplayTask(void *pvParameter) {
